# mammuth/workers/geonode_ingestion.py import asyncio, httpx from kafka import KafkaProducer from tenacity import retry, stop_after_attempt SOURCES = { "istat_comuni": "https://sdmx.istat.it/SDMXWS/rest/data/IT1,DF_COMUNI", "eurostat_lau": "https://gisco-services.ec.europa.eu/distribution/v2/lau", "insee_tahiti": "https://api.insee.fr/metadonnees/V1/geo/commune", } class GeoNodeWorker: def __init__(self, source_key, topic): self.source = SOURCES[source_key] self.producer = KafkaProducer(bootstrap_servers="redpanda:9092") self.topic = topic @retry(stop=stop_after_attempt(3)) async def fetch(self): async with httpx.AsyncClient(timeout=30) as c: return (await c.get(self.source)).json() async def run(self): data = await self.fetch() for record in GeoNodeNormalizer.normalize(data): self.producer.send(self.topic, value=record) async def main(): workers = [ GeoNodeWorker("istat_comuni", "geonode.italia"), GeoNodeWorker("eurostat_lau", "geonode.eu"), GeoNodeWorker("insee_tahiti", "geonode.global"), ] await asyncio.gather(*[w.run() for w in workers])
